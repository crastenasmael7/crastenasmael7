#include <iostream>
#include <fstream>
#include <string>
using namespace std;

int main()
{
    string text = "Hello Everybody";

    ofstream file("welcome.txt");
    file << text;
    file.close();

    int pos = text.find("body");

    if (pos != string::npos)
    {
        text.replace(pos, 4, "one");
    }

    ofstream outfile("welcome.txt");
    outfile << text;
    outfile.close();

    cout << text;

    return 0;
}
